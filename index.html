const SHEET_NAME = "Suscripciones";
const META_SHEET = "Configuracion_App";

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var metaSheet = ss.getSheetByName(META_SHEET);
  var services = null, customTemplates = null, offlineNotes = null, updatedAt = 0;
  if (metaSheet) {
    try { services = JSON.parse(metaSheet.getRange("B1").getValue() || "null"); } catch(err) {}
    try { customTemplates = JSON.parse(metaSheet.getRange("B2").getValue() || "null"); } catch(err) {}
    try { offlineNotes = JSON.parse(metaSheet.getRange("B3").getValue() || "null"); } catch(err) {}
    try { updatedAt = Number(metaSheet.getRange("B5").getValue()) || 0; } catch(err) {}
  }
  var sheet = ss.getSheetByName(SHEET_NAME);
  var accounts = [];
  if (sheet) {
    var rows = sheet.getDataRange().getDisplayValues();
    if (rows.length > 1) {
      var headers = rows[0];
      accounts = rows.slice(1).map(function(r) {
        var obj = {};
        headers.forEach(function(h, i) { obj[h] = r[i]; });
        return obj;
      });
    }
  }
  var jsonStr = JSON.stringify({ ok: true, accounts: accounts, services: services, customTemplates: customTemplates, offlineNotes: offlineNotes, updatedAt: updatedAt });
  if (e && e.parameter && e.parameter.callback) {
    return ContentService.createTextOutput(e.parameter.callback + '(' + jsonStr + ')')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(jsonStr)
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(15000);
  try {
    var payload = null;
    if (e && e.postData && e.postData.contents) {
      try { payload = JSON.parse(e.postData.contents); } catch(err1) {}
    }
    if (!payload && e && e.parameter && e.parameter.data) {
      try { payload = JSON.parse(e.parameter.data); } catch(err2) {}
    }
    if (!payload) {
      return ContentService.createTextOutput(JSON.stringify({ ok: false, error: "Sin datos" }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME);
    sheet.clearContents();
    var headers = ["id","service","email","pass","profile","pin","client","startDate","expiryDate","status","tag"];
    sheet.appendRow(headers);

    var accList = payload.accounts || [];
    if (accList.length > 0) {
      var values = accList.map(function(acc) {
        return headers.map(function(h) {
          return (acc[h] !== undefined && acc[h] !== null) ? String(acc[h]) : "";
        });
      });
      sheet.getRange(2, 1, values.length, headers.length).setValues(values);
    }

    var metaSheet = ss.getSheetByName(META_SHEET) || ss.insertSheet(META_SHEET);
    if (payload.services) {
      metaSheet.getRange("A1").setValue("SERVICES");
      metaSheet.getRange("B1").setValue(JSON.stringify(payload.services));
    }
    if (payload.customTemplates) {
      metaSheet.getRange("A2").setValue("TEMPLATES");
      metaSheet.getRange("B2").setValue(JSON.stringify(payload.customTemplates));
    }
    if (payload.offlineNotes) {
      metaSheet.getRange("A3").setValue("NOTES");
      metaSheet.getRange("B3").setValue(JSON.stringify(payload.offlineNotes));
    }
    var updateTimestamp = Number(payload.updatedAt) || Date.now();
    metaSheet.getRange("A5").setValue("UPDATED_AT");
    metaSheet.getRange("B5").setValue(String(updateTimestamp));

    return ContentService.createTextOutput(JSON.stringify({ ok: true, count: accList.length, updatedAt: updateTimestamp }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ ok: false, error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    try { lock.releaseLock(); } catch(e) {}
  }
}
